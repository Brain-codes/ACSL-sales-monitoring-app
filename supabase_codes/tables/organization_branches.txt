// SQL DEFINITION

CREATE TABLE public.organization_branches (
  id UUID NOT NULL DEFAULT gen_random_uuid(),
  organization_id UUID NOT NULL,
  name TEXT NOT NULL,
  country TEXT NOT NULL DEFAULT 'Nigeria',
  state TEXT NULL,
  lga TEXT NULL,
  created_at TIMESTAMP WITH TIME ZONE NULL DEFAULT now(),
  updated_at TIMESTAMP WITH TIME ZONE NULL DEFAULT now(),
  created_by UUID NOT NULL,
  
  CONSTRAINT organization_branches_pkey PRIMARY KEY (id),
  CONSTRAINT organization_branches_organization_id_fkey 
    FOREIGN KEY (organization_id) 
    REFERENCES public.organizations(id) 
    ON DELETE CASCADE,
  CONSTRAINT organization_branches_created_by_fkey 
    FOREIGN KEY (created_by) 
    REFERENCES auth.users(id) 
    ON DELETE RESTRICT,
  CONSTRAINT organization_branches_name_org_unique 
    UNIQUE (organization_id, name)
) TABLESPACE pg_default;

-- Create indexes for better performance
CREATE INDEX idx_organization_branches_org_id ON public.organization_branches(organization_id);
CREATE INDEX idx_organization_branches_country ON public.organization_branches(country);
CREATE INDEX idx_organization_branches_state ON public.organization_branches(state);
CREATE INDEX idx_organization_branches_created_at ON public.organization_branches(created_at);

-- Update trigger for updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_organization_branches_updated_at 
    BEFORE UPDATE ON public.organization_branches 
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

// RLS POLICIES

-- Enable RLS
ALTER TABLE public.organization_branches ENABLE ROW LEVEL SECURITY;

-- Super admin can do everything
CREATE POLICY "Super admin full access" ON public.organization_branches
  FOR ALL 
  USING (
    EXISTS (
      SELECT 1 FROM public.profiles 
      WHERE profiles.id = auth.uid() 
      AND profiles.role = 'super_admin'
    )
  );

-- Organization admins can manage their organization's branches
CREATE POLICY "Organization admin access" ON public.organization_branches
  FOR ALL 
  USING (
    EXISTS (
      SELECT 1 FROM public.profiles p
      JOIN public.organizations o ON p.organization_id = o.id
      WHERE p.id = auth.uid() 
      AND p.role = 'admin'
      AND o.id = organization_branches.organization_id
    )
  );

-- Agents can read their organization's branches
CREATE POLICY "Agent read access" ON public.organization_branches
  FOR SELECT 
  USING (
    EXISTS (
      SELECT 1 FROM public.profiles p
      JOIN public.organizations o ON p.organization_id = o.id
      WHERE p.id = auth.uid() 
      AND p.role = 'agent'
      AND o.id = organization_branches.organization_id
    )
  );

// TABLE PURPOSE
This table stores branch information for organizations. Each organization can have multiple branches in different locations. Branches include location details like country (default Nigeria), state, and LGA (Local Government Area) for Nigerian locations.

// RELATIONSHIPS
- organization_id: References organizations(id) with CASCADE delete
- created_by: References auth.users(id) with RESTRICT delete

// CONSTRAINTS
- Each branch name must be unique within an organization
- organization_id and created_by are required
- country defaults to 'Nigeria'
